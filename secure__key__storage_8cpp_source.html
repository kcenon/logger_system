36struct current_user {
37 std::vector<uint8_t> storage;
38
39 bool query() {
40 HANDLE raw = nullptr;
41 if (!OpenThreadToken(GetCurrentThread(), TOKEN_QUERY, TRUE, &raw)) {
42 if (GetLastError() != ERROR_NO_TOKEN ||
43 !OpenProcessToken(GetCurrentProcess(), TOKEN_QUERY, &raw)) {
44 return false;
45 }
46 }
47 native_handle token(raw);
48 DWORD size = 0;
49 GetTokenInformation(token.get(), TokenUser, nullptr, 0, &size);
50 if (size == 0) {
51 return false;
52 }
53 storage.resize(size);
54 return GetTokenInformation(token.get(), TokenUser, storage.data(), size, &size) != FALSE;
55 }
56
57 PSID sid() const {
58 return reinterpret_cast<const TOKEN_USER*>(storage.data())->User.Sid;
59 }
60};

62bool has_private_dacl(HANDLE file, PSID user) {
63 PACL acl = nullptr;
64 PSECURITY_DESCRIPTOR raw = nullptr;
65 const auto status = GetSecurityInfo(file, SE_FILE_OBJECT, DACL_SECURITY_INFORMATION,
66 nullptr, nullptr, &acl, nullptr, &raw);
67 local_buffer descriptor(raw);
68 // A null DACL grants access to everyone, unlike an empty DACL.
69 if (status != ERROR_SUCCESS || !acl || !IsValidAcl(acl)) {
70 return false;
71 }
72 bool can_read = false;
73 for (DWORD index = 0; index < acl->AceCount; ++index) {
74 void* entry = nullptr;
75 if (!GetAce(acl, index, &entry)) {
76 return false;
77 }
78 const auto* header = static_cast<const ACE_HEADER*>(entry);
79 if (header->AceType == ACCESS_DENIED_ACE_TYPE) {
80 continue;
81 }
82 // Fail closed for other principals and unrecognized ACE forms.
83 if (header->AceType != ACCESS_ALLOWED_ACE_TYPE) {
84 return false;
85 }
86 auto* allowed = static_cast<ACCESS_ALLOWED_ACE*>(entry);
87 if (!EqualSid(&allowed->SidStart, user)) {
88 return false;
89 }
90 can_read |= (allowed->Mask & (FILE_READ_DATA | GENERIC_READ | GENERIC_ALL)) != 0;
91 }
92 return can_read;
93}

97common::VoidResult secure_key_storage::save_key_windows(
98 const secure_key& key, const std::filesystem::path& path) {
99 current_user user;
100 if (!user.query()) {
101 return make_logger_void_result(logger_error_code::file_permission_denied,
102 "Failed to identify the Windows key owner");
103 }
104
105 EXPLICIT_ACCESSW access{};
106 access.grfAccessPermissions = FILE_ALL_ACCESS;
107 access.grfAccessMode = SET_ACCESS;
108 access.grfInheritance = NO_INHERITANCE;
109 access.Trustee.TrusteeForm = TRUSTEE_IS_SID;
110 access.Trustee.TrusteeType = TRUSTEE_IS_USER;
111 access.Trustee.ptstrName = reinterpret_cast<LPWSTR>(user.sid());
112 PACL raw_acl = nullptr;
113 const auto acl_status = SetEntriesInAclW(1, &access, nullptr, &raw_acl);
114 local_buffer acl(raw_acl);
115 SECURITY_DESCRIPTOR descriptor{};
116 if (acl_status != ERROR_SUCCESS ||
117 !InitializeSecurityDescriptor(&descriptor, SECURITY_DESCRIPTOR_REVISION) ||
118 !SetSecurityDescriptorDacl(&descriptor, TRUE, raw_acl, FALSE) ||
119 !SetSecurityDescriptorControl(&descriptor, SE_DACL_PROTECTED, SE_DACL_PROTECTED)) {
120 return make_logger_void_result(logger_error_code::file_permission_denied,
121 "Failed to create a private Windows key DACL");
122 }
123
124 SECURITY_ATTRIBUTES attributes{sizeof(SECURITY_ATTRIBUTES), &descriptor, FALSE};
125 native_handle file(CreateFileW(path.c_str(), GENERIC_WRITE | READ_CONTROL | WRITE_DAC,
126 0, &attributes, OPEN_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr));
127 if (file.get() == INVALID_HANDLE_VALUE) {
128 return make_logger_void_result(logger_error_code::file_write_failed,
129 "Failed to open the Windows key file for writing");
130 }
131 // Creation uses the private DACL atomically; existing files are restricted
132 // before any key bytes are written. Do not inherit the directory's grants.
133 if (SetSecurityInfo(file.get(), SE_FILE_OBJECT,
134 DACL_SECURITY_INFORMATION | PROTECTED_DACL_SECURITY_INFORMATION,
135 nullptr, nullptr, raw_acl, nullptr) != ERROR_SUCCESS ||
136 !has_private_dacl(file.get(), user.sid())) {
137 return make_logger_void_result(logger_error_code::file_permission_denied,
138 "Failed to enforce the Windows key DACL");
139 }
140
141 DWORD written = 0;
142 if (key.size() > std::numeric_limits<DWORD>::max() || !SetEndOfFile(file.get()) ||
143 !WriteFile(file.get(), key.data().data(), static_cast<DWORD>(key.size()), &written, nullptr) ||
144 written != key.size() || !FlushFileBuffers(file.get())) {
145 return make_logger_void_result(logger_error_code::file_write_failed,
146 "Failed to write the Windows key file");
147 }
148 return common::ok();
149}

151result<secure_key> secure_key_storage::load_key_windows(
152 const std::filesystem::path& path, size_t expected_size) {
153 native_handle file(CreateFileW(path.c_str(), GENERIC_READ | READ_CONTROL, FILE_SHARE_READ,
154 nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr));
155 if (file.get() == INVALID_HANDLE_VALUE) {
156 return {logger_error_code::file_open_failed, "Failed to open the Windows key file"};
157 }
158 current_user user;
159 if (!user.query() || !has_private_dacl(file.get(), user.sid())) {
160 return {logger_error_code::insecure_permissions,
161 "Windows key DACL must grant access only to the current user"};
162 }
163
164 LARGE_INTEGER size{};
165 if (!GetFileSizeEx(file.get(), &size)) {
166 return {logger_error_code::file_read_failed, "Failed to get the Windows key file size"};
167 }
168 if (expected_size > std::numeric_limits<DWORD>::max() ||
169 size.QuadPart != static_cast<LONGLONG>(expected_size)) {
170 return {logger_error_code::invalid_key_size, "Invalid Windows key file size"};
171 }
172
173 secure_key key(expected_size);
174 DWORD read = 0;
175 // Read from the same handle whose DACL and size were verified.
176 if (!ReadFile(file.get(), key.mutable_data().data(), static_cast<DWORD>(expected_size),
177 &read, nullptr) || read != expected_size) {
178 return {logger_error_code::file_read_failed, "Failed to read the Windows key file"};
179 }
180 return result<secure_key>(std::move(key));
181}
