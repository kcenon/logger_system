47inline ssize_t safe_write(int fd, const void* buf, size_t count) {
48 // MSVC terminates on a negative descriptor instead of returning EBADF.
49 if (fd < 0) {
50 errno = EBADF;
51 return -1;
52 }
53#ifdef _WIN32
54 return _write(fd, buf, static_cast<unsigned int>(count));
55#else
56 return ::write(fd, buf, count);
57#endif
58}

208 static void signal_handler(int sig) {
209 // Write signal message using only signal-safe functions
210 const char* msg = nullptr;
211 size_t msg_len = 0;
212
213 switch (sig) {
214 case SIGSEGV:
215 msg = "\n[CRITICAL] SIGSEGV received - emergency flush\n";
216 msg_len = 45;
217 break;
218 case SIGABRT:
219 msg = "\n[CRITICAL] SIGABRT received - emergency flush\n";
220 msg_len = 45;
221 break;
222 case SIGTERM:
223 msg = "\n[CRITICAL] SIGTERM received - emergency flush\n";
224 msg_len = 45;
225 break;
226 case SIGINT:
227 msg = "\n[CRITICAL] SIGINT received - emergency flush\n";
228 msg_len = 44;
229 break;
230 default:
231 msg = "\n[CRITICAL] Unknown signal - emergency flush\n";
232 msg_len = 43;
233 break;
234 }
235
236 // Write to stderr (signal-safe)
237 detail::safe_write(STDERR_FILENO, msg, msg_len);
238
239 // Emergency flush for all registered loggers
240 // Note: We cannot safely lock mutex_ here (not signal-safe)
241 // Instead, we access the loggers directly (risk of race condition,
242 // but in a crash scenario, this is acceptable)
243 signal_manager* mgr = current_instance_.load(std::memory_order_acquire);
244 if (!mgr) {
245 // No instance available, nothing to flush
246 _exit(128 + sig);
247 }
248
249 // Call emergency_flush on each logger
250 // Note: This assumes the logger pointers are still valid
251 // In production, consider using a signal-safe data structure
252 for (auto* log : mgr->loggers_) {
253 if (log) {
254 emergency_flush(log);
255 }
256 }
257
258 // Call original handler (or terminate)
259 switch (sig) {
260 case SIGSEGV:
261 if (mgr->original_sigsegv_ != SIG_DFL &&
262 mgr->original_sigsegv_ != SIG_IGN) {
263 mgr->original_sigsegv_(sig);
264 } else {
265 std::signal(SIGSEGV, SIG_DFL);
266 std::raise(SIGSEGV);
267 }
268 break;
269 case SIGABRT:
270 if (mgr->original_sigabrt_ != SIG_DFL &&
271 mgr->original_sigabrt_ != SIG_IGN) {
272 mgr->original_sigabrt_(sig);
273 } else {
274 std::signal(SIGABRT, SIG_DFL);
275 std::raise(SIGABRT);
276 }
277 break;
278 case SIGTERM:
279 if (mgr->original_sigterm_ != SIG_DFL &&
280 mgr->original_sigterm_ != SIG_IGN) {
281 mgr->original_sigterm_(sig);
282 } else {
283 _exit(128 + sig);
284 }
285 break;
286 case SIGINT:
287 if (mgr->original_sigint_ != SIG_DFL &&
288 mgr->original_sigint_ != SIG_IGN) {
289 mgr->original_sigint_(sig);
290 } else {
291 _exit(128 + sig);
292 }
293 break;
294 default:
295 _exit(128 + sig);
296 }
297 }

353// Implementation of emergency_flush (signal-safe)
354inline void signal_manager::emergency_flush(critical_logger_interface* log) {
355 if (!log) {
356 return;
357 }
358
359 // Get file descriptor
360 int fd = log->get_emergency_fd();
361 if (fd < 0) {
362 return;
363 }
364
365 // Get buffer
366 const char* buffer = log->get_emergency_buffer();
367 size_t size = log->get_emergency_buffer_size();
368
369 if (buffer && size > 0) {
370 // Write using signal-safe write() syscall
371 ssize_t written = 0;
372 while (written < static_cast<ssize_t>(size)) {
373 ssize_t n = detail::safe_write(fd, buffer + written, size - written);
374 if (n <= 0) {
375 break; // Error or would block
376 }
377 written += n;
378 }
379
380 // Sync to disk (signal-safe)
381 detail::safe_fsync(fd);
382 }
383}
